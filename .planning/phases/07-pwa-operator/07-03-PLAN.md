Create src/lib/ingest/notifying-adapter.ts — a decorator that `implements IngestAdapter`, wraps an inner adapter + the db + ctx (tenantId, plantId, plantName), delegates every method verbatim, and after `reportStop` fires the notifier:
```ts
import type { IngestAdapter, StopType } from './interface'
import type { BetterSQLite3Database } from 'drizzle-orm/better-sqlite3'
import * as schema from '@/db/schema'
import { notifyStop } from '@/lib/notifier/notifier'

export class NotifyingAdapter implements IngestAdapter {
  constructor(
    private inner: IngestAdapter,
    private db: BetterSQLite3Database<typeof schema>,
    private ctx: { tenantId: number; plantId: number; plantName: string },
  ) {}

  reportReading(...a: Parameters<IngestAdapter['reportReading']>) { return this.inner.reportReading(...a) }
  reportStopEnded(...a: Parameters<IngestAdapter['reportStopEnded']>) { return this.inner.reportStopEnded(...a) }
  reportBale(...a: Parameters<IngestAdapter['reportBale']>) { return this.inner.reportBale(...a) }
  ensureShift(...a: Parameters<IngestAdapter['ensureShift']>) { return this.inner.ensureShift(...a) }
  flush() { return this.inner.flush() }

  reportStop(plantId: number, startAt: Date, reason: string, stopType: StopType): number {
    const id = this.inner.reportStop(plantId, startAt, reason, stopType)
    // fire-and-forget — never let a notify failure break the synchronous ingest path
    void notifyStop(this.db, {
      tenantId: this.ctx.tenantId, plantId, plantName: this.ctx.plantName,
      stopId: id, reason, stopType, startAtMs: startAt.getTime(),
    }).catch((e) => console.error('[notifier] notifyStop failed', e))
    return id
  }
}
```
Note: reportStop in IngestAdapter is synchronous and returns a number — keep it synchronous; the notify is fire-and-forget (`void ... .catch`). The notifier's internal throttle prevents the per-minute re-open storm.
