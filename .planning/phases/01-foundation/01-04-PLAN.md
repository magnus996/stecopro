  <action>
Create `src/lib/nav.ts` as the single source of truth for navigation, typed against `UserRole` from `@/db/schema`:
```typescript
import type { UserRole } from '@/db/schema'

export type NavItem = { label: string; href: string; roles: UserRole[] }

export const NAV_ITEMS: NavItem[] = [
  { label: 'Dashbord',       href: '/dashboard', roles: ['operator', 'produksjonsleder', 'admin', 'system_admin'] },
  { label: 'Skiftrapporter', href: '/reports/shifts', roles: ['operator', 'produksjonsleder', 'admin', 'system_admin'] },
  { label: 'Analyser',       href: '/reports', roles: ['produksjonsleder', 'admin', 'system_admin'] },
  { label: 'Anleggsoppsett', href: '/admin/plant', roles: ['produksjonsleder', 'admin', 'system_admin'] },
  { label: 'Brukere',        href: '/admin/users', roles: ['admin', 'system_admin'] },
  { label: 'Tenants',        href: '/admin/tenants', roles: ['system_admin'] },
]

export function navItemsForRole(role: UserRole): NavItem[] {
  return NAV_ITEMS.filter((item) => item.roles.includes(role))
}
```
These labels reflect the Norwegian UI and the role access defined in TENA-03 (operator: dashboards/shift reports; produksjonsleder: all reports + plant config; admin: user management; system_admin: tenant management). The target routes (except /dashboard) are placeholders that later phases fill in.

Mark the items that point at not-yet-built routes is unnecessary — links are fine even if the route 404s in Phase 1 (only /dashboard exists yet).

Create `src/components/LogoutButton.tsx` — a client component with a form whose action is the `logout` server action from `@/actions/auth` (a button "Logg ut").
